///|
fn relay(value : String) -> String {
  value
}

///|
fn html_text(value : String) -> String {
  "<p>" + value + "</p>"
}

///|
async test "security-chain raw query reaches HTML response" {
  let app = @mocket.new()
  app.get("/review", event => {
    let value = event.req.query().get("q").unwrap_or("")
    @mocket.html(html_text(relay(value)))
  })
  let response = @mocket.dispatch_http(
    app, "GET", "/review?q=%3Cimg%20src%3Dx%20onerror%3Dalert%281%29%3E", {}, b"",
  )
  assert_eq(response.headers.get("Content-Type"), Some("text/html; charset=utf-8"))
  let actual : String = response.read_body()
  assert_eq(actual, "<p><img src=x onerror=alert(1)></p>")
  println("SECURITY_CHAIN raw " + actual)
}

///|
async test "security-chain escaped query is encoded in HTML text" {
  let app = @mocket.new()
  app.get("/review", event => {
    let value = event.req.query().get("q").unwrap_or("")
    @mocket.html(html_text(@mocket.escape_html(relay(value))))
  })
  let response = @mocket.dispatch_http(
    app, "GET", "/review?q=%3Cimg%20src%3Dx%20onerror%3Dalert%281%29%3E", {}, b"",
  )
  assert_eq(response.headers.get("Content-Type"), Some("text/html; charset=utf-8"))
  let actual : String = response.read_body()
  assert_eq(actual, "<p>&lt;img src=x onerror=alert(1)&gt;</p>")
  println("SECURITY_CHAIN escaped " + actual)
}

///|
async test "security-chain HTML escaping is not JavaScript neutralization" {
  let app = @mocket.new()
  app.get("/review", event => {
    let value = event.req.query().get("q").unwrap_or("")
    @mocket.html("<script>" + @mocket.escape_html(relay(value)) + "</script>")
  })
  let response = @mocket.dispatch_http(app, "GET", "/review?q=alert%281%29", {}, b"")
  let actual : String = response.read_body()
  assert_eq(actual, "<script>alert(1)</script>")
  println("SECURITY_CHAIN script_context " + actual)
}
