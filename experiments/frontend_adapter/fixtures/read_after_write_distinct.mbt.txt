pub struct Box { mut value : String; mut safe : String }
fn source() -> String { "dirty" }
fn sink(value : String) -> Unit { ignore(value) }
fn fill(first : Box, second : Box) -> Unit {
  first.value = source()
  second.safe = second.value
}
pub fn go() -> Unit {
  let a = Box::{value: "clean", safe: "clean"}
  let b = Box::{value: "clean", safe: "clean"}
  fill(a, b)
  sink(a.value)
  sink(a.safe)
  sink(b.value)
  sink(b.safe)
}
